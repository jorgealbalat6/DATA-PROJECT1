# Parsear la URL de conexión
url = os.getenv("DATABASE_URL")
result = urllib.parse.urlparse(url)
username = result.username
password = result.password
database = result.path[1:]
hostname = result.hostname
port = result.port

jdbc_url = f"jdbc:postgresql://{hostname}:{port}/{database}"

df = spark.read \
    .format("jdbc") \
    .option("url", jdbc_url) \
    .option("dbtable", '"marts_CalidadAire"') \
    .option("user", username) \
    .option("password", password) \
    .option("driver", "org.postgresql.Driver") \
    .option("fetchsize", "10000") \
    .load()

print(f"Datos cargados exitosamente en Spark: {df.count()} filas")
print("Esquema del DataFrame:")
df.printSchema()
print("Valores distintos en 'indicador':")
df.select("indicador").distinct().show()

df_pm25 = df.filter(col("indicador").isin("PM2.5", "9"))
df_pm10 = df.filter(col("indicador").isin("PM10", "10"))
df_no2 = df.filter(col("indicador").isin("NO2", "8"))
df_o3 = df.filter(col("indicador").isin("O3", "14"))
df_so2 = df.filter(col("indicador").isin("SO2", "1"))
df_co = df.filter(col("indicador").isin("CO", "6"))

In [6]:
cuantiles = df_pm25.approxQuantile("valor", [0.25, 0.5, 0.75], 0.01)

q1 = cuantiles[0]
mediana = cuantiles[1]
q3 = cuantiles[2]

riq = q3 - q1
bigote_bajo = q1 - (1.5 * riq)
bigote_alto = q3 + (1.5 * riq)

estadisticos = [{
    'med': mediana,
    'q1': q1,
    'q3': q3,
    'whislo': bigote_bajo,  
    'whishi': bigote_alto,
    'label': 'Boxplot PM2.5',
    'fliers': []
}]

fig, ax = plt.subplots()
ax.bxp(estadisticos, showfliers=False) 

plt.title('Boxplot PM2.5')
plt.grid(True)
plt.show()

barras = 20

bordes_intervalos, conteo_intervalos = df_pm25.select('valor').rdd.map(lambda x: float(x[0]) if x[0] is not None else 0.0).histogram(barras)

ancho_barras = bordes_intervalos[1] - bordes_intervalos[0]
centros_barras = [b + ancho_barras / 2 for b in bordes_intervalos[:-1]]

plt.bar(centros_barras, conteo_intervalos, width=ancho_barras, color='salmon', edgecolor='black')
plt.title(f'Histograma PM2.5 ({barras} barras)')
plt.xlabel('Valor')
plt.ylabel('Conteo')
plt.show()
